```c

{
    int i, max = 0;   /**** 错误 1 ****/
    for (i = 1; i < n; i++)
        if (a[i] > max)
            max = a[i];
    return max;
}

int main()
{
    int arr[5] = {-3, -1, -5, -2, -4};
    printf("%d\n", findMax(arr, 5));
    return 0;
}
```
